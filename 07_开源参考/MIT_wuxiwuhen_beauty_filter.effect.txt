uniform float4x4 ViewProj;
uniform texture2d image;
uniform texture2d low_tex;
uniform float2 uv_pixel;
uniform float smooth_strength;
uniform float white_strength;

sampler_state textureSampler {
	Filter    = Linear;
	AddressU  = Clamp;
	AddressV  = Clamp;
};

struct VertData {
	float4 pos : POSITION;
	float2 uv  : TEXCOORD0;
};

VertData VSDefault(VertData v_in)
{
	VertData vert_out;
	vert_out.pos = mul(float4(v_in.pos.xyz, 1.0), ViewProj);
	vert_out.uv  = v_in.uv;
	return vert_out;
}

// Gaussian 5-tap blur: two cascaded rounds of sigma=8.8px each, effective sigma ~= 12.
// Weights exp(-k^2/2) k=0,1,2 normalized: 0.40260 / 0.24419 / 0.05449; offsets 0/8.8/17.6 px.
// BlurH1/BlurH2 share PSBlurH, BlurV1/BlurV2 share PSBlurV.

float4 PSBlurH(VertData v_in) : TARGET
{
	float2 o = float2(uv_pixel.x, 0.0);
	float4 c = image.Sample(textureSampler, v_in.uv) * 0.40260;
	c += image.Sample(textureSampler, v_in.uv + o * 8.8) * 0.24419;
	c += image.Sample(textureSampler, v_in.uv - o * 8.8) * 0.24419;
	c += image.Sample(textureSampler, v_in.uv + o * 17.6) * 0.05449;
	c += image.Sample(textureSampler, v_in.uv - o * 17.6) * 0.05449;
	return c;
}

float4 PSBlurV(VertData v_in) : TARGET
{
	float2 o = float2(0.0, uv_pixel.y);
	float4 c = image.Sample(textureSampler, v_in.uv) * 0.40260;
	c += image.Sample(textureSampler, v_in.uv + o * 8.8) * 0.24419;
	c += image.Sample(textureSampler, v_in.uv - o * 8.8) * 0.24419;
	c += image.Sample(textureSampler, v_in.uv + o * 17.6) * 0.05449;
	c += image.Sample(textureSampler, v_in.uv - o * 17.6) * 0.05449;
	return c;
}

// smoothstep expanded: t = saturate((x-a)/(b-a)); t*t*(3-2t)

float SoftWindow(float x, float lo, float hi, float band)
{
	float t0 = saturate((x - (lo - band)) / (2.0 * band));
	float s0 = t0 * t0 * (3.0 - 2.0 * t0);
	float t1 = saturate((x - (hi - band)) / (2.0 * band));
	float s1 = t1 * t1 * (3.0 - 2.0 * t1);
	return s0 * (1.0 - s1);
}

// BT.601 full-range YCbCr, Cb/Cr centered at +0.5; windows CB 77..127, CR 133..173 (255 scale)

float SkinMask(float3 rgb)
{
	float cb = -0.168736 * rgb.r - 0.331264 * rgb.g + 0.5 * rgb.b + 0.5;
	float cr = 0.5 * rgb.r - 0.418688 * rgb.g - 0.081312 * rgb.b + 0.5;
	float band = 0.03;
	float cbm = SoftWindow(cb, 0.301961, 0.498039, band);
	float crm = SoftWindow(cr, 0.521569, 0.678431, band);
	return cbm * crm;
}

float4 PSComposite(VertData v_in) : TARGET
{
	float4 orig = image.Sample(textureSampler, v_in.uv);
	float3 low = low_tex.Sample(textureSampler, v_in.uv).rgb;

	float mask = SkinMask(orig.rgb);

	// Luma Sobel (ksize=3, unnormalized), 8-neighborhood
	float3 LUMA = float3(0.299, 0.587, 0.114);
	float2 px = uv_pixel;
	float l00 = dot(image.Sample(textureSampler, v_in.uv + float2(-px.x, -px.y)).rgb, LUMA);
	float l01 = dot(image.Sample(textureSampler, v_in.uv + float2(0.0, -px.y)).rgb, LUMA);
	float l02 = dot(image.Sample(textureSampler, v_in.uv + float2(px.x, -px.y)).rgb, LUMA);
	float l10 = dot(image.Sample(textureSampler, v_in.uv + float2(-px.x, 0.0)).rgb, LUMA);
	float l12 = dot(image.Sample(textureSampler, v_in.uv + float2(px.x, 0.0)).rgb, LUMA);
	float l20 = dot(image.Sample(textureSampler, v_in.uv + float2(-px.x, px.y)).rgb, LUMA);
	float l21 = dot(image.Sample(textureSampler, v_in.uv + float2(0.0, px.y)).rgb, LUMA);
	float l22 = dot(image.Sample(textureSampler, v_in.uv + float2(px.x, px.y)).rgb, LUMA);
	float gx = (l02 + 2.0 * l12 + l22) - (l00 + 2.0 * l10 + l20);
	float gy = (l20 + 2.0 * l21 + l22) - (l00 + 2.0 * l01 + l02);
	float edge = saturate(sqrt(gx * gx + gy * gy) * 10.0);

	float3 high = orig.rgb - low;
	float atten = mask * (1.0 - edge) * smooth_strength;
	float3 result = low + high * (1.0 - atten);

	// gamma whitening (WHITE_GAMMA_MIN = 0.52), lerp expanded
	float gamma = 1.0 + (0.52 - 1.0) * white_strength;
	float3 lifted = pow(saturate(result), float3(gamma, gamma, gamma));
	result = result + (lifted - result) * (mask * white_strength);

	return float4(result, orig.a);
}

technique BlurH1
{
	pass
	{
		vertex_shader = VSDefault(v_in);
		pixel_shader  = PSBlurH(v_in);
	}
}

technique BlurV1
{
	pass
	{
		vertex_shader = VSDefault(v_in);
		pixel_shader  = PSBlurV(v_in);
	}
}

technique BlurH2
{
	pass
	{
		vertex_shader = VSDefault(v_in);
		pixel_shader  = PSBlurH(v_in);
	}
}

technique BlurV2
{
	pass
	{
		vertex_shader = VSDefault(v_in);
		pixel_shader  = PSBlurV(v_in);
	}
}

technique Draw
{
	pass
	{
		vertex_shader = VSDefault(v_in);
		pixel_shader  = PSComposite(v_in);
	}
}
